local function addResult(severity, category, title, description, details)
	table.insert(PentestResults[severity], {

local function truncateStr(str, maxLen)
	maxLen = maxLen or 100
	if type(str) ~= "string" then str = tostring(str) end
	if #str > maxLen then
		return str:sub(1, maxLen) .. "..."
	end
	return str
end

local function getFullPath(instance)
	if not instance then return "nil" end
	local ok, path = pcall(function() return instance:GetFullName() end)
	if ok then return path end
	return tostring(instance)
end

local function isGameRemote(instance)
	if not instance then return false end
	local path = getFullPath(instance)
	if path:find("RobloxReplicatedStorage") then return false end
	if path:find("CoreGui") then return false end
	return true
end

local function generateTypeConfusionPayloads()
	return {
		{ name = "nil value", value = nil },
		{ name = "wrong type: number as string", value = "999999999" },
		{ name = "wrong type: string as number", value = 0 },
		{ name = "negative number", value = -999999 },
		{ name = "very large number", value = math.huge },
		{ name = "NaN", value = 0/0 },
		{ name = "empty string", value = "" },
		{ name = "very long string", value = string.rep("A", 10000) },
		{ name = "special chars string", value = "<script>alert(1)</script>" },
		{ name = "nested empty table", value = {{}} },
		{ name = "table with mixed keys", value = {[1] = "a", ["key"] = "b"} },
		{ name = "boolean true", value = true },
		{ name = "boolean false", value = false },
	}
end

-- SCANNER 1: RemoteEvent Scanner
-- ============================================================
local function scanRemoteEvents()
	addResult("info", "Scanner", "RemoteEvent Scan", "Memindai semua RemoteEvent yang bisa diakses client...")
	
	local found = {}
	
	local function scanContainer(container)
		if not container then return end
		local ok, descendants = pcall(function() return container:GetDescendants() end)
		if not ok then return end
		
		for _, desc in descendants do
			if desc:IsA("RemoteEvent") and isGameRemote(desc) then
				table.insert(found, desc)
				addResult("info", "RemoteEvent", "Ditemukan: " .. desc.Name, 
					"RemoteEvent ditemukan di: " .. getFullPath(desc))
			end
		end
	end
	
	scanContainer(ReplicatedStorage)
	scanContainer(Workspace)
	scanContainer(Lighting)
	scanContainer(StarterGui)
	scanContainer(StarterPlayer)
	scanContainer(StarterPack)
	
	if PlayerGui then
		scanContainer(PlayerGui)
	end
	local backpack = LocalPlayer:FindFirstChild("Backpack")
	if backpack then
		scanContainer(backpack)
	end
	
	PentestResults.stats.remoteEventsFound = #found
	
	if #found == 0 then
		addResult("passed", "RemoteEvent", "Tidak ada RemoteEvent kustom", 
			"Tidak ditemukan RemoteEvent kustom yang bisa diakses client.")
	else
		addResult("warning", "RemoteEvent", #found .. " RemoteEvent ditemukan", 
			"Ditemukan " .. #found .. " RemoteEvent. Pastikan semua memiliki validasi server.")
	end
	
	return found
end

-- SCANNER 3: Validation Tester
-- ============================================================
local function testValidation(remoteEvents)
	addResult("info", "Validation", "Validation Test", "Menguji apakah RemoteEvent memiliki validasi server...")
	
	if #remoteEvents == 0 then
		addResult("passed", "Validation", "Tidak ada RemoteEvent untuk ditest",
			"Tidak ada RemoteEvent kustom yang bisa diuji validasinya.")
		return
	end
	
	for _, remote in remoteEvents do
		local path = getFullPath(remote)
		
		local ok, err = pcall(function()
			remote:FireServer()
		end)
		PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
		
		if ok then
			addResult("warning", "Validation", "FireServer() tanpa argumen diterima: " .. remote.Name,
				"RemoteEvent '" .. path .. "' menerima pemanggilan tanpa argumen. " ..
				"Server mungkin tidak memvalidasi jumlah argumen.")
		else
			addResult("passed", "Validation", "FireServer() tanpa argumen ditolak: " .. remote.Name,
				"RemoteEvent menolak pemanggilan tanpa argumen: " .. truncateStr(err))
		end
		
		if not CONFIG.SAFE_MODE then
			local ok2 = pcall(function()
				remote:FireServer("BLOXCRYPT_PENTEST", nil, {}, -999999)
			end)
			PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
			
			if ok2 then
				addResult("critical", "Validation", "Argumen tidak valid diterima: " .. remote.Name,
					"RemoteEvent '" .. path .. "' menerima argumen campuran tanpa error. " ..
					"Kemungkinan besar tidak ada validasi tipe data di server.")
			end
		end
	end
end

-- SCANNER 4: Data Exposure Scanner
-- ============================================================
local function scanDataExposure()
	addResult("info", "DataExposure", "Data Exposure Scan", "Memindai data sensitif yang terekspos di client...")
	
	local ok, repDescendants = pcall(function() return ReplicatedStorage:GetDescendants() end)
	if ok then
		for _, desc in repDescendants do
			if desc:IsA("ModuleScript") then
				addResult("warning", "DataExposure", "ModuleScript di ReplicatedStorage: " .. desc.Name,
					"ModuleScript '" .. getFullPath(desc) .. "' bisa diakses client. " ..
					"Pastikan tidak berisi logika server-only atau rahasia.")
			end
			
			if desc:IsA("ValueBase") then
				local val = ""
				pcall(function() val = truncateStr(tostring(desc.Value), 50) end)
				addResult("warning", "DataExposure", "Value object terekspos: " .. desc.Name,
					"Object '" .. getFullPath(desc) .. "' (" .. desc.ClassName .. ") bisa dibaca client. " ..
					"Nilai: " .. val)
			end
		end
	end
	
	local ok2, wsDescendants = pcall(function() return Workspace:GetDescendants() end)
	if ok2 then
		for _, desc in wsDescendants do
			if desc:IsA("Configuration") or (desc:IsA("Folder") and string.lower(desc.Name):find("config")) then
				addResult("warning", "DataExposure", "Konfigurasi di Workspace: " .. desc.Name,
					"Folder konfigurasi ditemukan di Workspace yang bisa diakses client: " .. getFullPath(desc))
			end
		end
	end
	
	PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
end

-- ============================================================
-- SCANNER 5: Client Authority Scanner
-- ============================================================
local function scanClientAuthority()
	addResult("info", "ClientAuthority", "Client Authority Scan", "Memindai objek dengan otoritas client berlebihan...")
	
	local ok, descendants = pcall(function() return Workspace:GetDescendants() end)
	if not ok then return end
	
	local unanchoredCount = 0
	local networkOwnerParts = 0
	
	for _, desc in descendants do
		if desc:IsA("BasePart") then
			if not desc.Anchored then
				unanchoredCount = unanchoredCount + 1
			end
			
			if not desc.CanCollide and desc:IsA("Part") and not desc.Anchored then
				local ok3, size = pcall(function() return desc.Size end)
				if ok3 and size.Magnitude > 5 then
					addResult("info", "ClientAuthority", "Part non-collide: " .. desc.Name,
						"Part '" .. getFullPath(desc) .. "' tidak anchored dan CanCollide=false. " ..
						"Client bisa menembus part ini.")
				end
			end
		end
	end
	
	if unanchoredCount > 0 then
		addResult("info", "ClientAuthority", unanchoredCount .. " part tidak anchored",
			"Ditemukan " .. unanchoredCount .. " part yang tidak anchored di Workspace. " ..
			"Client bisa memanipulasi physics part ini.")
	end
	
	PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
end

-- SCANNER 6: Race Condition Tester
-- ============================================================
local function testRaceConditions(remoteEvents)
	addResult("info", "RaceCondition", "Race Condition Test", "Menguji kondisi race pada RemoteEvent...")
	
	if #remoteEvents == 0 then
		addResult("passed", "RaceCondition", "Tidak ada RemoteEvent untuk ditest",
			"Tidak ada RemoteEvent kustom yang bisa diuji race condition-nya.")
		return
	end
	
	if CONFIG.SAFE_MODE then
		addResult("info", "RaceCondition", "Race condition test dilewati (SAFE_MODE)",
			"Set SAFE_MODE = false untuk menguji race condition. " ..
			"PERINGATAN: Test ini bisa menyebabkan efek samping pada game.")
		return
	end
	
	for _, remote in remoteEvents do
		local path = getFullPath(remote)
		
		local ok = pcall(function()
			for i = 1, CONFIG.RACE_TEST_COUNT do
				remote:FireServer("RACE_TEST_" .. i, i)
				if CONFIG.RACE_TEST_DELAY > 0 then
					task.wait(CONFIG.RACE_TEST_DELAY)
				end
			end
		end)
		PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
		
		if ok then
			addResult("warning", "RaceCondition", "Race test berhasil dikirim: " .. remote.Name,
				"Berhasil mengirim " .. CONFIG.RACE_TEST_COUNT .. " request cepat ke '" .. path .. "'. " ..
				"Jika server memproses semua tanpa debounce/lock, ada potensi race condition.")
		end
	end
end

-- SCANNER 7: Type Confusion Tester
-- ============================================================
local function testTypeConfusion(remoteEvents)
	addResult("info", "TypeConfusion", "Type Confusion Test", "Menguji RemoteEvent dengan tipe data salah...")
	
	if #remoteEvents == 0 then
		addResult("passed", "TypeConfusion", "Tidak ada RemoteEvent untuk ditest",
			"Tidak ada RemoteEvent kustom yang bisa diuji type confusion-nya.")
		return
	end
	
	if CONFIG.SAFE_MODE then
		addResult("info", "TypeConfusion", "Type confusion test dilewati (SAFE_MODE)",
			"Set SAFE_MODE = false untuk menguji type confusion. " ..
			"PERINGATAN: Test ini bisa menyebabkan error pada server.")
		return
	end
	
	local payloads = generateTypeConfusionPayloads()
	
	for _, remote in remoteEvents do
		local path = getFullPath(remote)
		local acceptedCount = 0
		
		for _, payload in payloads do
			local ok = pcall(function()
				remote:FireServer(payload.value)
			end)
			PentestResults.stats.testsRun = PentestResults.stats.testsRun + 1
			
			if ok then
				acceptedCount = acceptedCount + 1
			end
		end
		
		if acceptedCount > 0 then
			addResult("warning", "TypeConfusion", acceptedCount .. "/" .. #payloads .. " payload diterima: " .. remote.Name,
				"RemoteEvent '" .. path .. "' menerima " .. acceptedCount .. " dari " .. #payloads ..
				" tipe payload berbeda. Server mungkin tidak memvalidasi tipe data.")
		end
	end
end

			"Client bisa mengakses isi ServerStorage. Ini seharusnya tidak mungkin terjadi.")
	else
		addResult("passed", "PermissionBypass", "ServerStorage terlindungi",
			"Client tidak bisa mengakses ServerStorage. Aman.")
	end
	
	local ok3, spawn = pcall(function() return Workspace:FindFirstChild("SpawnLocation") end)

		local ok4, _ = pcall(function()

			spawn.Anchored = original
		end)
		if ok4 then
			addResult("warning", "PermissionBypass", "SpawnLocation bisa dimodifikasi client",
				"Client berhasil mengubah properti SpawnLocation. " ..
				"Perubahan ini tidak akan direplikasi ke server, tapi bisa menyebabkan inkonsistensi lokal.")
		end
	end
	
	local ok5, guiDescendants = pcall(function() return PlayerGui:GetDescendants() end)
	if ok5 then
		for _, desc in guiDescendants do
			if desc:IsA("ScreenGui") then
				local name = string.lower(desc.Name)
				if name:find("admin") or name:find("mod") or name:find("debug") or name:find("command") then
					addResult("warning", "PermissionBypass", "Admin/Debug GUI terekspos: " .. desc.Name,
						"GUI dengan nama mencurigakan ditemukan di PlayerGui: " .. desc.Name ..
						". Pastikan ini tidak bisa diakses oleh player biasa.")
				end
			end
		end
	end
	
	PentestResults.stats.testsRun = PentestResults.stats.testsRun + 4

local function refreshResultsDisplay()
	if not resultsScrollFrame then return end
	
	for _, child in ipairs(resultsScrollFrame:GetChildren()) do
		if child:IsA("Frame") and child.Name:match("^Result_") then
			child:Destroy()
		end
	end
	
	local severityColors = {
		critical = Color3.fromRGB(255, 50, 50),
		warning = Color3.fromRGB(255, 180, 0),
		info = Color3.fromRGB(100, 150, 255),
		passed = Color3.fromRGB(50, 255, 100),
	}
	
	local severityOrder = {critical = 1, warning = 2, info = 3, passed = 4}
	
	local allResults = {}
	for sev, items in pairs(PentestResults) do
		if type(items) == "table" and sev ~= "stats" then
			for _, item in ipairs(items) do
				item.severity = sev
				table.insert(allResults, item)
			end
		end
	end
	
	table.sort(allResults, function(a, b)
		return (severityOrder[a.severity] or 5) < (severityOrder[b.severity] or 5)
	end)
	
	for i, result in ipairs(allResults) do
		local row = Instance.new("Frame")
		row.Name = "Result_" .. i
		row.Size = UDim2.new(1, -10, 0, 50)
		row.Position = UDim2.new(0, 5, 0, 0)
		row.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
		row.BorderSizePixel = 0
		row.LayoutOrder = severityOrder[result.severity] or 5
		row.Parent = resultsScrollFrame
		
		local rowCorner = Instance.new("UICorner")
		rowCorner.CornerRadius = UDim.new(0, 4)
		rowCorner.Parent = row
		
		local indicator = Instance.new("Frame")
		indicator.Size = UDim2.new(0, 4, 0.8, 0)
		indicator.Position = UDim2.new(0, 5, 0.1, 0)
		indicator.BackgroundColor3 = severityColors[result.severity]
		indicator.BorderSizePixel = 0
		indicator.Parent = row

		indCorner.CornerRadius = UDim.new(0, 2)

		local title = Instance.new("TextLabel")

		title.BackgroundTransparency = 1

	table.insert(lines, string.format("Waktu Eksekusi:  %.2f detik", elapsed))
	table.insert(lines, "")
	table.insert(lines, "----------------------------------------")
	table.insert(lines, "DETAIL TEMUAN:")
	table.insert(lines, "")
	
	for _, sev in {"critical", "warning", "info", "passed"} do
		if #PentestResults[sev] > 0 then
			table.insert(lines, "--- " .. string.upper(sev) .. " ---")
			for _, result in ipairs(PentestResults[sev]) do
				table.insert(lines, string.format("[%s] %s", result.category, result.title))
				table.insert(lines, "    " .. result.description)
				table.insert(lines, "")
			end
		end
	end
	
	table.insert(lines, "========================================")
	table.insert(lines, "Pengujian selesai!")
	table.insert(lines, "========================================")
	
	return table.concat(lines, "\n")
end

local function copyToClipboard(text)
	local success = false
	local function syncCopy()
		local clipboardSuccess = false
		local clipboardFunctions = {setclipboard, toclipboard}
		for _, func in ipairs(clipboardFunctions) do
			if type(func) == "function" then
				pcall(function() func(text) end)
				clipboardSuccess = true
				break
			end
		end
		return clipboardSuccess
	end
	
	success = syncCopy()
	
	if not success then
		local frame = Instance.new("Frame")
		frame.Size = UDim2.new(0, 0, 0, 0)
		frame.Parent = screenGui or PlayerGui
		
		local textBox = Instance.new("TextBox")
		textBox.Size = UDim2.new(0, 0, 0, 0)
		textBox.Text = text
		textBox.Parent = frame
		
		task.wait()
		frame:Destroy()
		success = true
	end
	
	return success
end
